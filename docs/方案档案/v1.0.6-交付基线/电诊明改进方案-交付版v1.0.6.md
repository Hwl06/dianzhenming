async function loadEnv() {
  const file = join(root, '.env');
  if (!existsSync(file)) return;
  const txt = await readFile(file, 'utf8');
  for (const line of txt.split(/\r?\n/)) {
    const s = line.trim();
    if (!s || s.startsWith('#')) continue;
    const i = s.indexOf('=');
    if (i < 0) continue;
    const k = s.slice(0, i).trim();
    let v = s.slice(i + 1).trim();
    if ((v.startsWith('"') && v.endsWith('"')) || (v.startsWith("'") && v.endsWith("'"))) v = v.slice(1, -1);
    if (!(k in process.env)) process.env[k] = v;
  }
}

function log(level, msg, extra) {
  const line = `[${new Date().toISOString()}] [${level}] ${msg}${extra ? ' ' + JSON.stringify(extra) : ''}`;
  if (level !== 'debug' || logLevel === 'debug') console.log(line);

const historyFile = join(dataDir, 'history.jsonl');
async function appendHistory(entry) {
  await appendFile(historyFile, JSON.stringify(entry) + '\n').catch(() => {});
  try {
    const txt = await readFile(historyFile, 'utf8');
    const lines = txt.trim().split('\n');
    if (lines.length > historyMax) {
      await writeFile(historyFile, lines.slice(-historyMax).join('\n') + '\n');
    }
  } catch {}
}
async function readHistory(limit = 20) {
  try {
    const txt = await readFile(historyFile, 'utf8');
    const lines = txt.trim().split('\n').filter(Boolean);
    return lines.slice(-limit).map(l => { try { return JSON.parse(l); } catch { return null; } }).filter(Boolean).reverse();
  } catch { return []; }
}

// ---------- 限流 ----------
const hits = new Map();
function rateLimited(ip) {
  const now = Date.now();
  const arr = (hits.get(ip) || []).filter(t => now - t < RATE_WINDOW);
  arr.push(now);
  hits.set(ip, arr);
  if (hits.size > 2000) for (const [k, v] of hits) if (!v.some(t => now - t < RATE_WINDOW)) hits.delete(k);
  return arr.length > RATE_LIMIT;
}

const MEASURE_KEYS = ['currentA','currentB','currentC','voltageA','voltageB','voltageC','temperature','ambient','load','extra'];
const SAFETY_KEYS = ['poweredOff','verified','locked','hazard'];
const RISK_SET = new Set([RISK.LOW, RISK.MEDIUM, RISK.HIGH]);

function validInput(x) {
  if (!x || typeof x !== 'object') return false;
  for (const k of ['device','symptom','alarm']) if (typeof (x[k] ?? '') !== 'string') return false;
  if (x.symptom.trim().length === 0 || x.symptom.length > 1000) return false;
  if (typeof x.measure !== 'object' || x.measure === null) return false;
  for (const k of MEASURE_KEYS) {
    const v = x.measure[k];
    if (v !== undefined && (typeof v !== 'string' || v.length > 200)) return false;
  }
  if (typeof x.safety !== 'object' || x.safety === null) return false;
  for (const k of SAFETY_KEYS) if (typeof x.safety[k] !== 'boolean') return false;
  return true;
}

// ---------- AI ----------
function prompt(x) {
  const m = x.measure || {};
  const p = [];
  if (m.currentA || m.currentB || m.currentC) p.push(`三相电流 A/B/C: ${m.currentA || '-'}/${m.currentB || '-'}/${m.currentC || '-'} A`);
  if (m.voltageA || m.voltageB || m.voltageC) p.push(`三相电压 A/B/C: ${m.voltageA || '-'}/${m.voltageB || '-'}/${m.voltageC || '-'} V`);
  if (m.temperature) p.push(`设备温度: ${m.temperature}℃`);
  if (m.ambient) p.push(`环境温度: ${m.ambient}℃`);
  if (m.load) p.push(`负载率: ${m.load}%`);
  if (m.extra) p.push(`补充: ${m.extra}`);
  const safety = `断电=${x.safety.poweredOff} 验电=${x.safety.verified} 挂牌=${x.safety.locked} 高危迹象=${x.safety.hazard}`;
  return `你是电气设备故障诊断助手。只做辅助判断，不替代持证电工。根据以下现场信息，用中文返回严格 JSON：{"summary":"一句初步判断","causes":[{"text":"原因","evidence":["证据"]}],"steps":[{"text":"步骤","safety":"注意"}],"safety":"安全提醒","escalation":"停机或升级条件","riskLevel":"low|medium|high","confidence":0.0}。设备：${x.device}；现象：${x.symptom}；报警：${x.alarm || '无'}；${p.join('；') || '未填写测量值'}。安全确认：${safety}。禁止编造测量值；信息不足就明确说需要补充。`;
}
function safeParse(s) { try { return JSON.parse(s); } catch { return null; } }

function validateModelOutput(o) {
  if (!o || typeof o !== 'object') return null;
  const clean = {
    source: 'ai',
    riskLevel: RISK_SET.has(o.riskLevel) ? o.riskLevel : RISK.MEDIUM,
    summary: typeof o.summary === 'string' ? o.summary.slice(0, 500) : '',
    confidence: typeof o.confidence === 'number' && o.confidence >= 0 && o.confidence <= 1 ? o.confidence : 0.6,
    missingInfo: [], causes: [], steps: [],
    safety: typeof o.safety === 'string' ? o.safety.slice(0, 500) : '',
    escalation: typeof o.escalation === 'string' ? o.escalation.slice(0, 500) : '',
    references: [], timestamp: new Date().toISOString()
  };
  if (!clean.summary || !clean.safety || !clean.escalation) return null;
  if (Array.isArray(o.causes)) {
    clean.causes = o.causes.slice(0, 8).map(c => ({
      text: String(c?.text || c || '').slice(0, 300),
      evidence: Array.isArray(c?.evidence) ? c.evidence.slice(0, 5).map(e => String(e).slice(0, 200)) : []
    })).filter(c => c.text);
  }
  if (Array.isArray(o.steps)) {
    clean.steps = o.steps.slice(0, 10).map(s => ({
      text: String(s?.text || s || '').slice(0, 300),
      safety: typeof s?.safety === 'string' ? s.safety.slice(0, 200) : ''
    })).filter(s => s.text);
  }
  if (clean.causes.length === 0 || clean.steps.length === 0) return null;
  return clean;
}

// ---------- 诊断主流程 ----------
async function handleDiagnose(req, res, ip) {
  if (rateLimited(ip)) return error(res, ERR.RATE_LIMIT);

  let body = '', size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size > MAX_BODY) return error(res, ERR.TOO_LARGE);
    body += chunk;
  }
  const input = safeParse(body);
  if (!input) return error(res, ERR.BAD_JSON);
  if (!validInput(input)) return error(res, ERR.BAD_INPUT);

  const t0 = Date.now();
  let result;

  if (apiKey) {
    try {
      const upstream = await fetch('https://api.baiduqianfan.ai/v1/chat/completions', {
        method: 'POST',
        headers: { 'content-type': 'application/json', authorization: `Bearer ${apiKey}` },
        body: JSON.stringify({
          model,
          messages: [
            { role: 'system', content: '你输出可靠、保守、结构化的电气安全诊断。' },
            { role: 'user', content: prompt(input) }
          ],
          temperature: 0.1
        }),
        signal: AbortSignal.timeout(20_000)
      });
      if (!upstream.ok) {
        const msg = upstream.status === 401 ? 'API Key 无效'
          : upstream.status === 429 ? '模型服务限流'
          : `千帆接口返回 ${upstream.status}`;
        result = { ...ruleDiagnose(input, 'rule'), fallbackReason: `${msg}，已回退本地规则` };
        log('warn', 'ai fallback', { status: upstream.status });
      } else {
        const data = await upstream.json().catch(() => null);
        const raw = data?.choices?.[0]?.message?.content || '';
        const match = raw.match(/\{[\s\S]*\}/);
        const clean = match ? validateModelOutput(safeParse(match[0])) : null;
        result = clean || { ...ruleDiagnose(input, 'rule'), fallbackReason: '模型结果不符合格式，已回退本地规则' };
      }
    } catch (e) {
      result = { ...ruleDiagnose(input, 'rule'), fallbackReason: '模型服务不可用，已回退本地规则' };
      log('warn', 'ai exception', { err: e?.message });
    }
  } else {
    result = ruleDiagnose(input, 'rule');
  }

  const elapsed = Date.now() - t0;
  await appendHistory({
    ts: result.timestamp,
    source: result.source,
    riskLevel: result.riskLevel,
    device: input.device,
    symptom: input.symptom.slice(0, 120),
    elapsed
  });
  log('info', 'diagnose', { source: result.source, risk: result.riskLevel, elapsed });

  return ok(res, result);
}

const server = http.createServer(async (req, res) => {
  try {
    const ip = String(req.headers['x-forwarded-for'] || '').split(',')[0].trim() || req.socket.remoteAddress || 'unknown';
    const url = (req.url || '/').split('?')[0];

    if (req.method === 'POST' && url === '/api/diagnose') return await handleDiagnose(req, res, ip);

    if (req.method === 'GET' && url === '/health') {
      return ok(res, {
        ok: true,
        version: VERSION,
        aiEnabled: Boolean(apiKey),
        mode: apiKey ? 'online' : 'local',
        uptime: Math.round(process.uptime()),
        historyFile: existsSync(historyFile)
      });
    }
    if (req.method === 'GET' && url === '/api/config') {
      return ok(res, { version: VERSION, aiEnabled: Boolean(apiKey), model: apiKey ? model : null, historyMax });
    }
    if (req.method === 'GET' && url === '/api/history') {
      const u = new URL(req.url, 'http://x');
      const limit = Math.min(Number(u.searchParams.get('limit') || 20), 100);
      const items = await readHistory(limit);
      return ok(res, { items });
    }
    if (req.method === 'GET' && (await serveStatic(res, url))) return;

    res.writeHead(404, SECURITY); res.end('Not found');
  } catch (e) {
    log('error', 'server', { err: e?.message });
    error(res, ERR.INTERNAL);
  }
});
